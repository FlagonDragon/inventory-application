// Tables

CREATE TABLE champions (
   id INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
   name VARCHAR ( 30 )
);

CREATE TABLE weightClasses (
   id INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
   class VARCHAR ( 30 ) 
);

CREATE TABLE organizations (
   id INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
   fullName VARCHAR ( 50 ),
   acronym VARCHAR (5)
);

CREATE TABLE champions_organizations (
   id INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
   champion_id INTEGER,
   organization_id INTEGER 
);

CREATE TABLE champions_weightclasses (
   id INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
   champion_id INTEGER,
   weightclass_id INTEGER 
);
